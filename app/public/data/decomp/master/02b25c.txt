
void tz_auss_control(void)

{
  if (((((ZUSTAND_MOTOR & Nachlauf) == 0) &&
       (((ZUSTAND_MOTOR & KI.15 aus) == 0 || (TZ_KL15AUS_COUNT != '\0')))) &&
      (((SCHALT_M_ST & 8) == 0 || (K_TZ_NMIN_KL50ON <= N)))) &&
     ((((SCHALT_M_ST & 8) != 0 || (K_TZ_NMIN_KL50OFF <= N)) &&
      ((TPU_ST_M & nicht sync) == in Ordnung)))) {
    if (((TZ_IOS & 0x80) == 0) && (TZ_KANALSTATUS != DAT_00ffe62e)) {
      FUN_0002b7f8();
    }
  }
  else if (TZ_KANALSTATUS != '\0') {
    FUN_0002b7d6();
  }
  if ((((K_TZ_AUSS_CFG & 0xf0) == 0x30) && (TZ_KANALSTATUS == DAT_00ffe62e)) &&
     ((TPU_ST_M & keine NW) == in Ordnung)) {
    TZ_AUSS_ST = TZ_AUSS_ST | 1;
  }
  else {
    TZ_AUSS_ST = TZ_AUSS_ST & 0xfe;
  }
  if (TZ_AUSS_ST == 1) {
    TZ_AUSS_ST = 3;
    FUN_0002b81c();
    TZ_AUSS_ABSCHALTFLAG = DAT_00ffe62e;
  }
  if ((TZ_AUSS_ST == 2) && (TZ_AUSS_ABSCHALTFLAG == '\0')) {
    TZ_AUSS_ST = 0;
  }
  return;
}

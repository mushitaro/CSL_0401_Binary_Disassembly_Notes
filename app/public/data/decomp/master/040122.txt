
void sys_task_scheduler_init(void)

{
  byte slot_index;
  struct_task_schedule_table_entry *task_schedule_table_entry;
  
                    /* Reset system timekeeping counters */
  SYS_TIMERTICS_COUNTER = 0;
  SYS_SEC_COUNTER = 0;
  TimerTics_To_Next_Sec = 1024;
                    /* Start at the first task schedule entry */
  task_schedule_table_entry = sys_task_schedule_table;
                    /* Initialize 8 entries in the task schedule table */
  slot_index = 8;
  do {
    task_schedule_table_entry->task_id = 0;
    task_schedule_table_entry->slot_index = slot_index;
    task_schedule_table_entry->countdown_timer_tics = 0;
    task_schedule_table_entry = task_schedule_table_entry + 1;
    slot_index = slot_index - 1;
  } while (0 < (char)slot_index);
  return;
}

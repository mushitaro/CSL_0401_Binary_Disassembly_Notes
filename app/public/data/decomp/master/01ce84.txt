
void ub_uext1m_init(void)

{
  ushort uVar1;
  
  UB_UEXT_HR_ST = 0;
  if ((UB_HR_ST & 2) != 0) {
    UB_UEXT_HR_ST = 4;
  }
  uVar1 = uext1m_ad;
  uVar1 = ub_uext1m_diag(uVar1);
  if ((UB_UEXT_HR_ST & 4) != 0) {
    UEXT1M = uVar1 << 6;
    UEXT1M_DPR = UEXT1M;
    UEXT2M_KORREKTUR = (short)(0x80000 / (uint)((int)(UEXT1M + 0x20) >> 6));
    UB_UEXT_HR_ST = UB_UEXT_HR_ST | 0x10;
    return;
  }
  UEXT1M = (ushort)K_UEXTM_ERSATZ << 8;
  UEXT1M_DPR = (ushort)K_UEXTM_ERSATZ << 8;
  UEXT2M_KORREKTUR = (short)(0x80000 / (uint)((int)((uint)K_UEXTM_ERSATZ * 0x100 + 0x20) >> 6));
  return;
}
